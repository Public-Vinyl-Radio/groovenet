cmake_minimum_required(VERSION 3.20)
project(groovenet_brick VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL REQUIRED IMPORTED_TARGET sdl2>=2.0.18 SDL2_ttf>=2.0.18)
pkg_check_modules(NET REQUIRED IMPORTED_TARGET libcurl json-c libpng libjpeg)
find_package(Threads REQUIRED)
option(BRICK_COVERAGE "Clang source coverage" OFF)
if(BRICK_COVERAGE)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "BRICK_COVERAGE requires Clang")
  endif()
  add_compile_options(-fprofile-instr-generate -fcoverage-mapping)
  add_link_options(-fprofile-instr-generate)
endif()
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
add_library(brick src/app.cpp src/live.cpp src/artwork.cpp src/renderer.cpp src/desktop.cpp)
target_include_directories(brick PUBLIC src)
target_link_libraries(brick PUBLIC PkgConfig::SDL PkgConfig::NET Threads::Threads)
add_executable(groovenet-brick src/main.cpp)
target_link_libraries(groovenet-brick PRIVATE brick)
include(CTest)
if(BUILD_TESTING)
  add_executable(brick-tests tests/tests.cpp tests/live_tests.cpp tests/failures.cpp src/app.cpp src/live.cpp src/artwork.cpp)
  target_include_directories(brick-tests PRIVATE src)
  target_link_libraries(brick-tests PRIVATE PkgConfig::SDL PkgConfig::NET Threads::Threads)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME brick-release COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_release.py)
  add_test(NAME brick-package COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_package.py)
  add_test(NAME brick-network COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_live.py $<TARGET_FILE:brick-tests>)
  add_test(NAME brick-tests COMMAND brick-tests)
  set_tests_properties(brick-tests PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy")
endif()
