cmake_minimum_required(VERSION 3.20)
project(brick_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(DEPS REQUIRED IMPORTED_TARGET sdl2>=2.0.18 libcurl libpng)
add_executable(brick-probe main.cpp)
target_link_libraries(brick-probe PRIVATE PkgConfig::DEPS)
target_compile_options(brick-probe PRIVATE -Wall -Wextra -Wpedantic)
option(PROBE_COVERAGE "Clang source coverage" OFF)
if(PROBE_COVERAGE)
  target_compile_options(brick-probe PRIVATE -fprofile-instr-generate -fcoverage-mapping)
  target_link_options(brick-probe PRIVATE -fprofile-instr-generate)
endif()

include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME probe-smoke COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_probe.py $<TARGET_FILE:brick-probe>)
  add_executable(probe-input-test test_input.cpp)
  target_link_libraries(probe-input-test PRIVATE PkgConfig::DEPS)
  target_compile_options(probe-input-test PRIVATE -Wall -Wextra -Wpedantic)
  if(PROBE_COVERAGE)
    target_compile_options(probe-input-test PRIVATE -fprofile-instr-generate -fcoverage-mapping)
    target_link_options(probe-input-test PRIVATE -fprofile-instr-generate)
  endif()
endif()
